const questions = window.questions || []; const $ = id => document.getElementById(id); let current = 0; let lives = 5; let score = 0; let answered = false; function showScreen(id) { document.querySelectorAll(".screen").forEach(screen => { screen.classList.remove("active"); }); $(id).classList.add("active"); } function updateStats() { $("progress").textContent = `Câu ${Math.min(current + 1, questions.length)}/${questions.length}`; $("lives").textContent = "♥ ".repeat(lives); } function startGame() { if (!questions.length) { alert("Chưa tìm thấy dữ liệu câu hỏi gốc."); return; } current = 0; lives = 5; score = 0; showScreen("quiz"); renderQuestion(); } function renderQuestion() { if (current >= questions.length || lives <= 0) { finishGame(); return; } answered = false; const q = questions[current]; $("category").textContent = q.category || ""; $("question").textContent = q.question || ""; $("feedback").className = "feedback"; $("feedback").textContent = ""; $("nextBtn").style.display = "none"; updateStats(); const answers = $("answers"); answers.innerHTML = ""; ["A", "B", "C", "D"].forEach((letter, index) => { const button = document.createElement("button"); button.className = "answer"; button.innerHTML = `${letter}`; button.querySelector("span:last-child").textContent = q.answers[index]; button.addEventListener("click", () => chooseAnswer(index, button)); answers.appendChild(button); }); const eButton = document.createElement("button"); eButton.className = "answer e"; eButton.innerHTML = 'EKO BIẾT LÀM'; eButton.addEventListener("click", openModal); answers.appendChild(eButton); } function chooseAnswer(index, button) { if (answered) return; answered = true; const q = questions[current]; const buttons = $("answers").querySelectorAll(".answer"); buttons.forEach(b => b.disabled = true); if (index === q.correct) { score++; button.classList.add("correct"); $("feedback").textContent = "Chính xác! " + (q.explanation || ""); $("feedback").className = "feedback show good"; } else { lives--; button.classList.add("wrong"); if (buttons[q.correct]) { buttons[q.correct].classList.add("correct"); } $("feedback").textContent = "Chưa chính xác. " + (q.explanation || ""); $("feedback").className = "feedback show bad"; } updateStats(); $("nextBtn").style.display = "inline-block"; } function nextQuestion() { if (!answered) return; current++; renderQuestion(); } function openModal() { if (answered) return; $("modalBackdrop").classList.add("show"); } function closeModal() { $("modalBackdrop").classList.remove("show"); } function resetGame() { closeModal(); current = 0; lives = 5; score = 0; showScreen("quiz"); renderQuestion(); } function skipQuestion() { closeModal(); current++; renderQuestion(); } function finishGame() { showScreen("result"); $("resultHeading").textContent = lives <= 0 ? "Đã hết mạng!" : "Hoàn thành thử thách!"; $("resultText").textContent = `Bạn trả lời đúng ${score}/${questions.length} câu.`; } function goHome() { closeModal(); showScreen("home"); } function toggleFullscreen() { if (!document.fullscreenElement) { document.documentElement.requestFullscreen?.(); } else { document.exitFullscreen?.(); } } $("startBtn").addEventListener("click", startGame); $("restartBtn").addEventListener("click", resetGame); $("resultHomeBtn").addEventListener("click", goHome); $("backBtn").addEventListener("click", goHome); $("nextBtn").addEventListener("click", nextQuestion); $("resetChoice").addEventListener("click", resetGame); $("skipChoice").addEventListener("click", skipQuestion); $("closeModal").addEventListener("click", closeModal); $("homeFull").addEventListener("click", toggleFullscreen); $("fullBtn").addEventListener("click", toggleFullscreen); $("modalBackdrop").addEventListener("click", event => { if (event.target === $("modalBackdrop")) closeModal(); });
